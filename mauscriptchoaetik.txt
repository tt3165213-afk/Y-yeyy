-- 2. HÀM QUẢN LÝ VŨ KHÍ
local function setupWeaponHitbox(tool)
    if not tool:IsA("Tool") then return end
    local handle = tool:WaitForChild("Handle", 2) or tool:FindFirstChildWhichIsA("BasePart")
    if handle and handle:IsA("BasePart") then
        if _G.HackEnabled then
            handle.Color = COLOR_GREEN
            handle.Transparency = 0.4 
            handle.Material = Enum.Material.ForceField
        else
            handle.Transparency = 0 -- Trả về độ trong suốt mặc định
            handle.Material = Enum.Material.Plastic
        end
    end
end

-- 3. HÀM QUẢN LÝ HITBOX NGƯỜI
local function applyHitbox(character)
    local hrp = character:WaitForChild("HumanoidRootPart", 10)
    if hrp and hrp:IsA("BasePart") then
        if _G.HackEnabled then
            hrp.Size = BOX_HITBOX_SIZE
            hrp.Color = COLOR_GREEN
            hrp.Transparency = 0.5
            hrp.Material = Enum.Material.ForceField
            hrp.CanCollide = false
        else
            hrp.Size = ORIGINAL_HITBOX_SIZE
            hrp.Transparency = 1 -- Ẩn khối vuông đi khi tắt
            hrp.Material = Enum.Material.Plastic
        end
    end
end

-- Xử lý cho bản thân
local function setupLocalPlayer()
    local function onChar(character)
        applyHitbox(character)
        character.ChildAdded:Connect(setupWeaponHitbox)
        local currentTool = character:FindFirstChildOfClass("Tool")
        if currentTool then setupWeaponHitbox(currentTool) end
    end
    LocalPlayer.CharacterAdded:Connect(onChar)
    if LocalPlayer.Character then onChar(LocalPlayer.Character) end
end

-- 4. HÀM XỬ LÝ ĐỊCH & DRAWING ESP
local function applyEnemyHack(player)
    if player == LocalPlayer then return end

    local box = Drawing.new("Square")
    box.Visible = false
    box.Color = COLOR_GREEN
    box.Thickness = 1.5
    box.Filled = false

    local text = Drawing.new("Text")
    text.Visible = false
    text.Color = COLOR_GREEN
    text.Size = 14
    text.Center = true
    text.Outline = true
    text.OutlineColor = Color3.fromRGB(0, 0, 0)

    local connection
    connection = game:GetService("RunService").RenderStepped:Connect(function()
        if not player or not player.Parent or not player.Character then
            box.Visible = false
            text.Visible = false
            if not player.Parent then
                box:Remove()
                text:Remove()
                connection:Disconnect()
            end
            return
        end

        local character = player.Character
        local hrp = character:FindFirstChild("HumanoidRootPart")
        local head = character:FindFirstChild("Head")

        if hrp and head and _G.HackEnabled then
            -- Khi bật hack: Cập nhật liên tục kích thước và ESP
            if hrp.Size ~= BOX_HITBOX_SIZE then applyHitbox(character) end
            
            local holdingWeapon = character:FindFirstChildOfClass("Tool")
            if holdingWeapon then setupWeaponHitbox(holdingWeapon) end

            local hrpPosition, isOnScreen = Camera:WorldToViewportPoint(hrp.Position)
            if isOnScreen then
                local headPosition = Camera:WorldToViewportPoint(head.Position + Vector3.new(0, 0.5, 0))
                local legPosition = Camera:WorldToViewportPoint(hrp.Position - Vector3.new(0, 3, 0))
                local height = math.abs(headPosition.Y - legPosition.Y)
                local width = height / 1.5

                box.Size = Vector2.new(width, height)
                box.Position = Vector2.new(hrpPosition.X - width / 2, hrpPosition.Y - height / 2)
                box.Visible = true

                local weaponName = holdingWeapon and holdingWeapon.Name or "Tay Không"
                text.Text = "[ " .. weaponName .. " ]"
                text.Position = Vector2.new(hrpPosition.X, hrpPosition.Y + (height / 2) + 5)
                text.Visible = true
            else
                box.Visible = false
                text.Visible = false
            end
        else
            -- Khi tắt hoặc không hồi sinh: Ẩn toàn bộ ESP
            box.Visible = false
            text.Visible = false
            if hrp and hrp.Size == BOX_HITBOX_SIZE then applyHitbox(character) end
        end
    end)
end

-- Khởi động hệ thống ban đầu

Players.PlayerAdded:Connect(applyEnemyHack)

-- Thay đổi thuộc tính của các vật thể vật lý thành màu trơn, phẳng
local function fixLagObject(obj)
    if obj:IsA("BasePart") and not obj:IsA("MeshPart") then
        obj.Material = Enum.Material.SmoothPlastic
        obj.Reflectance = 0
    elseif obj:IsA("Texture") or obj:IsA("Decal") then
        -- Ép các hình ảnh bề mặt về kích thước tối thiểu bằng cách đổi ID trống hoặc xóa để game tự render 1x1
        obj.Texture = "" 
    elseif obj:IsA("MeshPart") then
        obj.Material = Enum.Material.SmoothPlastic
        obj.Reflectance = 0
    end

    -- 2. XÓA CÁC HIỆU ỨNG HẠT VÀ ÁNH SÁNG PHỨC TẠP TRÊN VẬT THỂ
    if obj:IsA("ParticleEmitter") or obj:IsA("Trail") or obj:IsA("Smoke") or obj:IsA("Fire") or obj:IsA("Sparkles") then
        obj.Enabled = false
    end
end

-- Quét toàn bộ vật thể đang có sẵn trong game
for _, obj in ipairs(workspace:GetDescendants()) do
    fixLagObject(obj)
end

-- Tự động quét và xử lý các vật thể mới xuất hiện (Ví dụ chiêu thức mới tung ra)
workspace.DescendantAdded:Connect(fixLagObject)

-- 3. TỐI ƯU HÓA ÁNH SÁNG VÀ XÓA HIỆU ỨNG MÔI TRƯỜNG (LIGHTING)

lighting.FogStart = 0
lighting.FogEnd = 999999 -- Xóa sương mù nhìn xa hơn và đỡ lag
lighting.ShadowMapBlur = 0

-- Xóa các hiệu ứng lọc màu, nhòe hình, tia sáng mặt trời
local function clearLightingEffects()
    for _, effect in ipairs(lighting:GetChildren()) do
        if effect:IsA("BloomEffect") or effect:IsA("BlurEffect") or effect:IsA("ColorCorrectionEffect") or effect:IsA("SunRaysEffect") or effect:IsA("DepthOfFieldEffect") then
            effect.Enabled = false
        end
    end
end

clearLightingEffects()
lighting.ChildAdded:Connect(clearLightingEffects)
